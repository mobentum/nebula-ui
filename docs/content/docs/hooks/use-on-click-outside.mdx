`useOnClickOutside` invokes a handler whenever a `mousedown` or `touchstart` event happens outside the referenced element. Commonly used for closing menus, popovers, and drawers.

```tsx
import { useOnClickOutside } from '@mobentum/nebula-ui';

function Popover() {
  const ref = useRef<HTMLDivElement>(null);
  const [open, setOpen] = useState(false);

  useOnClickOutside(ref, () => setOpen(false));

  return <div ref={ref}>{/* … */}</div>;
}
```

## API

```ts
useOnClickOutside<T extends HTMLElement>(
  ref: RefObject<T | null>,
  handler: (event: MouseEvent | TouchEvent) => void,
): void
```

## Usage

Close a dropdown when the user clicks away:

```tsx
function Menu({ trigger }: { trigger: ReactNode }) {
  const ref = useRef<HTMLDivElement>(null);
  const [open, setOpen] = useState(false);

  useOnClickOutside(ref, () => setOpen(false));

  return (
    <div ref={ref}>
      <button onClick={() => setOpen((v) => !v)}>{trigger}</button>
      {open && <div className="absolute …">Items</div>}
    </div>
  );
}
```

## Notes

- Listens on `document` for both `mousedown` and `touchstart`.
- Events inside the referenced element are ignored via `element.contains(event.target)`.
- The handler and ref are wired through effects, so the latest values are always used.

---
title: Switch
description: A toggle switch for binary on/off states.
---

import { SwitchRoot, SwitchThumb } from '@mobentum/nebula-ui';

## Basic

```tsx
<SwitchRoot>
  <SwitchThumb />
</SwitchRoot>
<SwitchRoot defaultChecked>
  <SwitchThumb />
</SwitchRoot>
```

<Preview>
  <div className="flex flex-wrap items-center gap-4">
    <SwitchRoot>
      <SwitchThumb />
    </SwitchRoot>
    <SwitchRoot defaultChecked>
      <SwitchThumb />
    </SwitchRoot>
  </div>
</Preview>

import { InputGroupAddon, InputGroupRoot } from '@nebula/ui';

## Basic

```tsx
<InputGroupRoot>
  <InputGroupAddon>@</InputGroupAddon>
  <input className="w-full rounded-r-md border px-3 py-2 text-sm" placeholder="Username" />
</InputGroupRoot>
```

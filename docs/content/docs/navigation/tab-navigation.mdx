import { TabNavigation, TabNavigationList, TabNavigationRoot, TabNavigationTab } from '@mobentum/nebula-ui';

## Basic

```tsx
<TabNavigationRoot defaultValue="tab1">
  <TabNavigationList>
    <TabNavigationTab value="tab1">Tab 1</TabNavigationTab>
    <TabNavigationTab value="tab2">Tab 2</TabNavigationTab>
    <TabNavigationTab value="tab3">Tab 3</TabNavigationTab>
  </TabNavigationList>
</TabNavigationRoot>
```

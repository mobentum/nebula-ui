```tsx
  <MenubarItem>File</MenubarItem>
  <MenubarItem>Edit</MenubarItem>
  <MenubarItem>View</MenubarItem>
  <MenubarSeparator />
  <MenubarItem>Help</MenubarItem>
</MenubarRoot>
```

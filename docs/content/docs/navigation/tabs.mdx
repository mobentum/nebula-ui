---
title: Tabs
description: Tabbed navigation for switching between content panels.
---

import { Tabs, TabsList, TabsPanel, TabsRoot, TabsTab } from '@mobentum/nebula-ui';

## Basic

```tsx
<TabsRoot defaultValue="tab1">
  <TabsList>
    <TabsTab value="tab1">Tab 1</TabsTab>
    <TabsTab value="tab2">Tab 2</TabsTab>
    <TabsTab value="tab3">Tab 3</TabsTab>
  </TabsList>
  <TabsPanel value="tab1">
    <p className="text-sm text-muted-foreground">Content for tab 1.</p>
  </TabsPanel>
  <TabsPanel value="tab2">
    <p className="text-sm text-muted-foreground">Content for tab 2.</p>
  </TabsPanel>
  <TabsPanel value="tab3">
    <p className="text-sm text-muted-foreground">Content for tab 3.</p>
  </TabsPanel>
</TabsRoot>
```

---
title: Pagination
description: A pagination component for navigating through pages of content.
---

import { Pagination, PaginationNext, PaginationPage, PaginationPrevious, PaginationRoot } from '@nebula/ui';

## Basic

```tsx
<PaginationRoot>
  <PaginationPrevious>Previous</PaginationPrevious>
  <PaginationPage value="1">1</PaginationPage>
  <PaginationPage value="2">2</PaginationPage>
  <PaginationPage value="3">3</PaginationPage>
  <PaginationNext>Next</PaginationNext>
</PaginationRoot>
```

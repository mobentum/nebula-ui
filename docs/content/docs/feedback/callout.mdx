import { Callout, CalloutContent, CalloutDescription, CalloutIcon, CalloutRoot, CalloutTitle } from '@nebula/ui';

## Basic

```tsx
<CalloutRoot variant="info">
  <CalloutContent>
    <CalloutTitle>Heads up!</CalloutTitle>
    <CalloutDescription>This is an informational message.</CalloutDescription>
  </CalloutContent>
</CalloutRoot>
```

        <CalloutDescription>This is an informational message.</CalloutDescription>
      </CalloutContent>
    </CalloutRoot>
  </div>
</Preview>

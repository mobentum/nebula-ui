import { Progress, ProgressIndicator, ProgressLabel, ProgressRoot, ProgressTrack, ProgressValueLabel } from '@mobentum/nebula-ui';

## Determinate

```tsx
<ProgressRoot value={65}>
  <ProgressLabel>Upload progress</ProgressLabel>
  <ProgressTrack>
    <ProgressIndicator />
  </ProgressTrack>
  <ProgressValueLabel />
</ProgressRoot>
```

<Preview>
  <ProgressRoot value={65} className="max-w-sm">
    <ProgressLabel>Upload progress</ProgressLabel>
    <ProgressTrack>
      <ProgressIndicator />
    </ProgressTrack>
    <ProgressValueLabel />
  </ProgressRoot>
</Preview>

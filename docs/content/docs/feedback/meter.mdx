---
title: Meter
description: A gauge component for visualizing a value within a known range.
---

import { Meter, MeterIndicator, MeterLabel, MeterRoot, MeterTrack, MeterValueLabel } from '@mobentum/nebula-ui';

## Basic

```tsx
<MeterRoot value={75}>
  <MeterLabel>Storage</MeterLabel>
  <MeterTrack>
    <MeterIndicator />
  </MeterTrack>
  <MeterValueLabel />
</MeterRoot>
```

<Preview>
  <MeterRoot value={75} className="max-w-sm">
    <MeterLabel>Storage</MeterLabel>
    <MeterTrack>
      <MeterIndicator />
    </MeterTrack>
    <MeterValueLabel />
  </MeterRoot>
</Preview>

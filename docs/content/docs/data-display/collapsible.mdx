---
title: Collapsible
description: A toggleable panel that shows and hides content.
---

import { Collapsible, CollapsiblePanel, CollapsibleRoot, CollapsibleTrigger } from '@mobentum/nebula-ui';

## Basic

```tsx
<CollapsibleRoot>
  <CollapsibleTrigger>Toggle content</CollapsibleTrigger>
  <CollapsiblePanel>
    This content can be shown or hidden.
  </CollapsiblePanel>
</CollapsibleRoot>
```
